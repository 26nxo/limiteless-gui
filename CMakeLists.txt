cmake_minimum_required(VERSION 3.24)
project(limitless LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(LIMITLESS_BUILD_PREVIEW "Build the Windows Direct3D 11 preview" ${WIN32})
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/imgui")
add_library(imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp)
target_include_directories(imgui PUBLIC ${IMGUI_DIR})
add_library(limitless_gui STATIC src/limitless/renderer.cpp)
target_include_directories(limitless_gui PUBLIC include)
target_link_libraries(limitless_gui PUBLIC imgui)
if(LIMITLESS_BUILD_PREVIEW)
    if(NOT WIN32)
        message(FATAL_ERROR "The preview harness requires Windows and Direct3D 11.")
    endif()
    add_executable(limitless_preview WIN32 src/preview/main.cpp ${IMGUI_DIR}/backends/imgui_impl_win32.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
    target_link_libraries(limitless_preview PRIVATE limitless_gui d3d11 dxgi dwmapi shell32)
    target_compile_definitions(limitless_preview PRIVATE UNICODE _UNICODE)
    add_custom_command(TARGET limitless_preview POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:limitless_preview>/assets")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(limitless_interaction_tests tests/limitless_interactions.cpp)
    target_link_libraries(limitless_interaction_tests PRIVATE limitless_gui)
    target_compile_definitions(limitless_interaction_tests PRIVATE LIMITLESS_TEST_FONT="${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/PreviewSans.ttf")
    add_test(NAME limitless_interactions COMMAND limitless_interaction_tests)
endif()
