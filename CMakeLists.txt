cmake_minimum_required(VERSION 3.24)
project(timewarp_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(GUI_BUILD_PREVIEW "Build the Windows Direct3D 11 preview" ${WIN32})
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/imgui")
add_library(imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp)
target_include_directories(imgui PUBLIC ${IMGUI_DIR})
add_library(gui STATIC src/gui/gui.cpp)
target_include_directories(gui PUBLIC include)
target_link_libraries(gui PUBLIC imgui)
if(GUI_BUILD_PREVIEW)
    if(NOT WIN32)
        message(FATAL_ERROR "The preview harness requires Windows and Direct3D 11.")
    endif()
    add_executable(timewarp_preview WIN32 main.cpp ${IMGUI_DIR}/backends/imgui_impl_win32.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
    target_link_libraries(timewarp_preview PRIVATE gui d3d11 dxgi dwmapi)
    target_compile_definitions(timewarp_preview PRIVATE UNICODE _UNICODE)
    add_custom_command(TARGET timewarp_preview POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:timewarp_preview>/assets")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(gui_interaction_tests tests/interactions.cpp)
    target_link_libraries(gui_interaction_tests PRIVATE gui)
    target_compile_definitions(gui_interaction_tests PRIVATE GUI_TEST_FONT="${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/PreviewSans.ttf")
    add_test(NAME gui_interactions COMMAND gui_interaction_tests)
endif()
